<!DOCTYPE html>

      <em>em pour mettre en italique</em>,
      <strong>strong pour mettre en gras</strong>

    </div>

    <div>

      <ul>
        <li>UL = unordered list</li>

        <li>UL = unordered list</li>
      </ul>

        <li>OL = ordered list</li>
        <li>OL = ordered list</li>
      </ol>

        <thead>
          <tr>
            <th>Col 1</th>
            <th>Col 2</th>
            <th>Col 3</th>
          </tr>
        </thead>
        <tbody>
          <tr>
            <td rowspan="2">Row 1 Cell 1</td>
            <td>Row 1 Cell 2</td>
            <td>Row 1 Cell 3</td>
          </tr>
          <tr>
            <td>Row 2 Cell 2</td>
            <td>Row 2 Cell 3</td>

            <td colspan="3">Row 3 Cell 1</td>
          </tr>
