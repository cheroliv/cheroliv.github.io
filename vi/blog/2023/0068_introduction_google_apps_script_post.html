Giới thiệu

Google Apps Script (GAS) là một nền tảng phát triển tập lệnh đi kèm với nhiều dịch vụ của Google như Google Sheets, Google Docs và Google Forms. Nó cho phép tự động hóa các tác vụ, thêm các tính năng tùy chỉnh và tích hợp các dịch vụ của Google.

Trong bản ghi nhớ này, chúng ta sẽ khám phá các khái niệm chính của Google Apps Script thông qua các ví dụ cụ thể. Chúng ta sẽ đề cập đến các khái niệm như biến, vòng lặp, hàm, và nhiều nội dung khác, được điều chỉnh cho phù hợp với bối cảnh cụ thể của GAS.

Cấu trúc cơ bản trong Google Apps Script

Để bắt đầu, hãy cùng xem xét cấu trúc cơ bản của một script Google Apps.

/**
 * Structure de base en Google Apps Script
 */
function exempleStructureGAS() {
    // Votre code ici
    Logger.log("Hello, Google Apps Script!");
}

Kiểu và Biến trong Google Apps Script

Google Apps Script sử dụng JavaScript làm ngôn ngữ lập trình. Các kiểu dữ liệu và biến hoạt động tương tự như JavaScript tiêu chuẩn.

/**
 * Types et Variables en Google Apps Script
 */
function exempleTypesVariablesGAS() {
    // Votre code ici
    var nombre = 42;
    var texte = "Bonjour, Google Apps Script!";
    Logger.log(nombre);
    Logger.log(texte);
}

Các Phép toán và Toán tử trong Google Apps Script

Các phép toán và toán tử trong Google Apps Script tương tự như trong JavaScript.

/**
 * Opérations et Opérateurs en Google Apps Script
 */
function exempleOperationsGAS() {
    // Votre code ici
    var x = 10;
    var y = 5;
    var resultat = x + y;
    Logger.log(resultat);
}

Var, Let và Const trong Google Apps Script

Trong GAS, các khái niệm về`var`, let, et `const`được sử dụng để khai báo biến. Tuy nhiên, việc hiểu rõ sự khác biệt và hệ quả của chúng là rất quan trọng.

/**
 * Var, Let et Const en Google Apps Script
 */
function exempleVarLetConstGAS() {
    // Votre code ici
    // Utilisation de var
    var x = 10;
    if (true) {
        var x = 20; // La variable x est modifiée globalement
    }
    Logger.log(x);

    // Utilisation de let
    let y = 30;
    if (true) {
        let y = 40; // La variable y est limitée au bloc if
    }
    Logger.log(y);

    // Utilisation de const
    const z = 50;
    // z = 60; // Impossible de réassigner une constante
    Logger.log(z);
}

Cấu trúc Điều khiển trong Google Apps Script

Các cấu trúc điều khiển như`if`, else, while et `for`được sử dụng để quản lý luồng thực thi trong Google Apps Script.

/**
 * Structures de Contrôle en Google Apps Script
 */
function exempleStructuresControleGAS() {
    // Votre code ici
    var condition = true;

    if (condition) {
        Logger.log("La condition est vraie.");
    } else {
        Logger.log("La condition est fausse.");
    }

    var compteur = 0;
    while (compteur < 5) {
        Logger.log(compteur);
        compteur++;
    }

    for (var i = 0; i < 3; i++) {
        Logger.log(i);
    }
}

Tham số và Phạm vi trong Google Apps Script

Các tham số và scope đóng vai trò quan trọng trong việc phát triển các script trong GAS. Hãy hiểu cách chúng hoạt động để tránh các hành vi ngoài ý muốn.

/**
 * Paramètres et Scopes en Google Apps Script
 */
function exempleParametresScopesGAS(parametre) {
    // Votre code ici
    var variableGlobale = "Je suis global";

    function afficherParametre() {
        Logger.log(parametre);
    }

    afficherParametre();
    Logger.log(variableGlobale);
}

Hằng số, Biến và Chữ cái trong Google Apps Script

Hiểu sự khác biệt giữa`const`, var, et `let`là điều thiết yếu để sử dụng hiệu quả các biến trong Google Apps Script.

/**
 * Constantes, Variables, et Lettres en Google Apps Script
 */
function exempleConstantesVariablesLettresGAS() {
    // Votre code ici
    const constante = "Je ne change pas";
    Logger.log(constante);

    var variable = "Je peux changer";
    Logger.log(variable);

    let lettre = "Je peux aussi changer, mais seulement dans mon bloc";
    Logger.log(lettre);
}

Vòng lặp trong Google Apps Script

Các vòng lặp, chẳng hạn như`for` et while, là những thành phần thiết yếu để lặp qua các phần tử và thực hiện các thao tác lặp đi lặp lại.

/**
 * Boucles en Google Apps Script
 */
function exempleBouclesGAS() {
    // Votre code ici
    for (var i = 0; i < 3; i++) {
        Logger.log(i);
    }

    var condition = true;
    var compteur = 0;

    while (condition) {
        Logger.log("Tour de boucle");
        compteur++;
        if (compteur === 3) {
            condition = false;
        }
    }
}

Đối tượng trong Google Apps Script

Các đối tượng được sử dụng để cấu trúc dữ liệu. Trong GAS, nhiều đối tượng tích hợp giúp đơn giản hóa việc tương tác với các dịch vụ của Google.

/**
 * Objets en Google Apps Script
 */
function exempleObjetsGAS() {
    // Votre code ici
    var feuille = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var cellule = feuille.getRange("A1");
    cellule.setValue("Nouvelle valeur");
}

Hằng số, Biến và Chữ cái trong Google Apps Script

Hiểu sự khác biệt giữa`const`, var, et `let`là điều thiết yếu để sử dụng hiệu quả các biến trong Google Apps Script.

/**
 * Constantes, Variables, et Lettres en Google Apps Script
 */
function exempleConstantesVariablesLettresGAS() {
    // Votre code ici
    const constante = "Je ne change pas";
    Logger.log(constante);

    var variable = "Je peux changer";
    Logger.log(variable);

    let lettre = "Je peux aussi changer, mais seulement dans mon bloc";
    Logger.log(lettre);

Các hàm trong Google Apps Script

Các hàm là những thành phần cơ bản trong lập trình. Trong GAS, chúng có thể được khai báo và gọi theo nhiều cách khác nhau.

/**
 * Fonctions en Google Apps Script
 */
function exempleFonctionsGAS() {
    // Votre code ici
    function additionner(a, b) {
        return a + b;
    }

    var resultat = additionner(2, 3);
    Logger.log(resultat);

    // Fonction anonyme
    var multiplier = function (x, y) {
        return x * y;
    };

    Logger.log(multiplier(4, 5));
}

Toán tử Nghỉ và Lan truyền trong Google Apps Script

Các toán tử nghỉ (…​) và lan truyền (…​) rất hữu ích để thao tác với mảng và đối tượng một cách súc tích.

/**
 * Opérateurs de Repos et de Propagation en Google Apps Script
 */
function exempleOperateursReposPropagationGAS() {
    // Votre code ici
    // Opérateur de repos (...) pour les tableaux
    var nombres = [1, 2, 3, 4, 5];
    var [...copieNombres] = nombres;
    Logger.log(copieNombres);

    // Opérateur de propagation (...) pour les objets
    var objOriginal = { x: 1, y: 2 };
    var objClone = { ...objOriginal, z: 3 };
    Logger.log(objClone);
}

Gán giá trị bằng Phân rã trong Google Apps Script

Phân rã gán giá trị (destructuring assignment) là một tính năng mạnh mẽ cho phép trích xuất các giá trị từ đối tượng và mảng một cách súc tích.

/**
 * Affectation par Décomposition en Google Apps Script
 */
function exempleAffectationDecompositionGAS() {
    // Votre code ici
    var coordonnees = [3, 4];
    var [x, y] = coordonnees;
    Logger.log(x);
    Logger.log(y);

    var utilisateur = { nom: "John", age: 30 };
    var { nom, age } = utilisateur;
    Logger.log(nom);
    Logger.log(age);
}

Collections và Lập trình Khai báo trong Google Apps Script

Google Apps Script hỗ trợ các cấu trúc dữ liệu như mảng, cũng như các khái niệm lập trình hàm và lặp khai báo (><mệnh lệnh), chẳng hạn như`map`, reduce et `forEach`Vì bạn chưa cung cấp văn bản cần dịch, vui lòng gửi đoạn văn bản tiếng Pháp bạn muốn chuyển sang tiếng Việt. Tôi sẽ tuân thủ chính xác tất cả các yêu cầu về định dạng và bảo toàn các đoạn mã trong dấu backtick.

/**
 * Collections et Programmation Déclarative en Google Apps Script
 */
function exempleCollectionsFonctionnelleGAS() {
    // Votre code ici
    // Exemple de tableau (collection)
    var nombres = [1, 2, 3, 4, 5];

    // Utilisation de forEach
    nombres.forEach(function (nombre) {
        Logger.log(nombre);
    });

    // Utilisation de map
    var carresNombres = nombres.map(function (nombre) {
        return nombre * nombre;
    });
    Logger.log(carresNombres);

    // Utilisation de reduce
    var somme = nombres.reduce(function (acc, nombre) {
        return acc + nombre;
    }, 0);
    Logger.log(somme);
}
