ভূমিকা

Google Apps Script (GAS) হল Google এর বিভিন্ন পরিষেবা, যেমন Google Sheets, Google Docs, এবং Google Forms এর সাথে যুক্ত স্ক্রিপ্ট ডেভেলপমেন্ট প্ল্যাটফর্ম। এটি কাজগুলি স্বয়ংক্রিয়করণ করার, কাস্টম ফিচার যোগ করার এবং Google পরিষেবাগুলি একীভূত করার অনুমতি দেয়।

এই মেমোতে, আমরা বাস্তব উদাহরণের সাহায্যে Google Apps Scriptের মূল ধারণাগুলো অন্বেষণ করব। আমরা ভেরিয়েবল, লুপ, ফাংশন এবং অন্যান্য অনেক ধারণা নিয়ে আলোচনা করব, এবং তা GAS-এ নির্দিষ্ট संदर्भে অ্যাড্যাপ্ট করব।

মৌলিক কাঠামো Google Apps Script-এ

শুরু করার জন্য, আমরা একটি Google Apps স্ক্রিপ্টের মৌলিক কাঠামো পর্যালোচনা করি।

/**
 * Structure de base en Google Apps Script
 */
function exempleStructureGAS() {
    // Votre code ici
    Logger.log("Hello, Google Apps Script!");
}

টাইপস এবং ভেরিয়েবল গুগল অ্যাপস স্ক্রিপ্টে

Google Apps Script প্রোগ্রামিং ভাষা হিসেবে JavaScript ব্যবহার করে। ডেটা টাইপ এবং ভেরিয়েবল JavaScript-এর মতো কাজ করে।

/**
 * Types et Variables en Google Apps Script
 */
function exempleTypesVariablesGAS() {
    // Votre code ici
    var nombre = 42;
    var texte = "Bonjour, Google Apps Script!";
    Logger.log(nombre);
    Logger.log(texte);
}

Google Apps Script-এ অপারেশন ও অপারেটর

Google Apps Script-এ অপারেশন এবং অপারেটর JavaScript-এ যেমন একই।

/**
 * Opérations et Opérateurs en Google Apps Script
 */
function exempleOperationsGAS() {
    // Votre code ici
    var x = 10;
    var y = 5;
    var resultat = x + y;
    Logger.log(resultat);
}

Var, Let এবং Const Google Apps Script-এ

GAS-এ, ধারণাগুলো`var`, let, et `const`এইগুলো ভেরিয়েবল ঘোষণা করতে ব্যবহৃত হয়। তবে, তাদের পার্থক্য এবং প্রভাব বুঝা জরুরি।

/**
 * Var, Let et Const en Google Apps Script
 */
function exempleVarLetConstGAS() {
    // Votre code ici
    // Utilisation de var
    var x = 10;
    if (true) {
        var x = 20; // La variable x est modifiée globalement
    }
    Logger.log(x);

    // Utilisation de let
    let y = 30;
    if (true) {
        let y = 40; // La variable y est limitée au bloc if
    }
    Logger.log(y);

    // Utilisation de const
    const z = 50;
    // z = 60; // Impossible de réassigner une constante
    Logger.log(z);
}

Google Apps Script-এ কন্ট্রোল স্ট্রাকচার

নিয়ন্ত্রণ কাঠামো যেমন`if`, else, while et `for`এগুলো Google Apps Script-এ চালানোর প্রবাহ ব্যবস্থাপনা করতে ব্যবহার করা হয়।

/**
 * Structures de Contrôle en Google Apps Script
 */
function exempleStructuresControleGAS() {
    // Votre code ici
    var condition = true;

    if (condition) {
        Logger.log("La condition est vraie.");
    } else {
        Logger.log("La condition est fausse.");
    }

    var compteur = 0;
    while (compteur < 5) {
        Logger.log(compteur);
        compteur++;
    }

    for (var i = 0; i < 3; i++) {
        Logger.log(i);
    }
}

সেটিংস এবং scopes Google Apps Script-এ

প্যারামিটার এবং স্কোপ একটি গুরুত্বপূর্ণ ভূমিকা পালন.koren GASe স্ক্রিপ্ট উন্নয়নে। বুঝুন কী</tool_call> তারা কাজ.koren অপ্রত্যাশিত আচরণ বাঁচার জন্য

/**
 * Paramètres et Scopes en Google Apps Script
 */
function exempleParametresScopesGAS(parametre) {
    // Votre code ici
    var variableGlobale = "Je suis global";

    function afficherParametre() {
        Logger.log(parametre);
    }

    afficherParametre();
    Logger.log(variableGlobale);
}

ধ্রুবক, চল, এবং অক্ষর Google Apps Script-এ

বিহিত পার্থক্য বোঝা`const`, var, et `let`Google Apps Script-এ বেরিয়েবলের কার্যকর ব্যবহার জন্য প্রয়োজনীয় আছে.

/**
 * Constantes, Variables, et Lettres en Google Apps Script
 */
function exempleConstantesVariablesLettresGAS() {
    // Votre code ici
    const constante = "Je ne change pas";
    Logger.log(constante);

    var variable = "Je peux changer";
    Logger.log(variable);

    let lettre = "Je peux aussi changer, mais seulement dans mon bloc";
    Logger.log(lettre);
}

Google Apps Scriptে লুপ

লুপ,for et while, উপাদানগুলোতে পুনরাবৃত্তি করতে এবং পুনরাবৃত্তি কাজগুলো সম্পাদন করতে অপরিহার্য।

/**
 * Boucles en Google Apps Script
 */
function exempleBouclesGAS() {
    // Votre code ici
    for (var i = 0; i < 3; i++) {
        Logger.log(i);
    }

    var condition = true;
    var compteur = 0;

    while (condition) {
        Logger.log("Tour de boucle");
        compteur++;
        if (compteur === 3) {
            condition = false;
        }
    }
}

Google Apps Script-এ অবজেক্ট

বস্তুগুলো ডেটা গঠনে ব্যবহৃত হয়। GAS‑এ, অনেকটি অন্তর্ভুক্ত বস্তু Google পরিষেবদের সাথে ইন্টারঅ্যাকশন সহজ করে।

/**
 * Objets en Google Apps Script
 */
function exempleObjetsGAS() {
    // Votre code ici
    var feuille = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var cellule = feuille.getRange("A1");
    cellule.setValue("Nouvelle valeur");
}

ধ্রুবক, ভেরিয়েবল, এবং অক্ষর Google Apps Script‑এ

অন্তর বুঝা`const`, var, et `let`Google Apps Script‑এ ভেরিয়েবলকে কার্যকরভাবে ব্যবহার করতে এটি অপরিহার্য।

/**
 * Constantes, Variables, et Lettres en Google Apps Script
 */
function exempleConstantesVariablesLettresGAS() {
    // Votre code ici
    const constante = "Je ne change pas";
    Logger.log(constante);

    var variable = "Je peux changer";
    Logger.log(variable);

    let lettre = "Je peux aussi changer, mais seulement dans mon bloc";
    Logger.log(lettre);
}

Google Apps Script-এ ফাংশন

ফাংশনগুলো প্রোগ্রামিংয়ের মৌলিক উপাদান। GAS-এ, সেগুলো ভিন্ন ভিন্ন পদ্ধতিতে ঘোষণা এবং কল করা যায়।

/**
 * Fonctions en Google Apps Script
 */
function exempleFonctionsGAS() {
    // Votre code ici
    function additionner(a, b) {
        return a + b;
    }

    var resultat = additionner(2, 3);
    Logger.log(resultat);

    // Fonction anonyme
    var multiplier = function (x, y) {
        return x * y;
    };

    Logger.log(multiplier(4, 5));
}

Google Apps Script-এ রেস্ট এবং স্প্রেড অপারেটর

বিশ্রামের অপারেটর (…​) এবং প্রসারনের (…​) arrays এবং objects কে সংক্ষেপে ব্যবহার করা উপযোগী।

/**
 * Opérateurs de Repos et de Propagation en Google Apps Script
 */
function exempleOperateursReposPropagationGAS() {
    // Votre code ici
    // Opérateur de repos (...) pour les tableaux
    var nombres = [1, 2, 3, 4, 5];
    var [...copieNombres] = nombres;
    Logger.log(copieNombres);

    // Opérateur de propagation (...) pour les objets
    var objOriginal = { x: 1, y: 2 };
    var objClone = { ...objOriginal, z: 3 };
    Logger.log(objClone);
}

Google Apps Script-এ বিশ্লেষণによる বিন্যাস

বিঘ্নিতকরণের মাধ্যমে অ্যাসাইনমেন্ট একটি শক্তিশালী বৈশিষ্ট্য যা অবজেক্ট এবং অ্যারে থেকে মান সংক্ষিপ্তভাবে বের করার অনুমতি দেয়।

/**
 * Affectation par Décomposition en Google Apps Script
 */
function exempleAffectationDecompositionGAS() {
    // Votre code ici
    var coordonnees = [3, 4];
    var [x, y] = coordonnees;
    Logger.log(x);
    Logger.log(y);

    var utilisateur = { nom: "John", age: 30 };
    var { nom, age } = utilisateur;
    Logger.log(nom);
    Logger.log(age);
}

Google Apps Script-এ সংগ্রহ ও ঘোষণামূলক প্রোগ্রামিং

Google Apps Script ডেটা গঠন kuten অ্যারে এবং ফাংশনাল প্রোগ্রামিং ধারণা ও ঘোষণা-ভিত্তিক ইটারেশন (><impérative), যেমন`map`, reduce et forEach.

/**
 * Collections et Programmation Déclarative en Google Apps Script
 */
function exempleCollectionsFonctionnelleGAS() {
    // Votre code ici
    // Exemple de tableau (collection)
    var nombres = [1, 2, 3, 4, 5];

    // Utilisation de forEach
    nombres.forEach(function (nombre) {
        Logger.log(nombre);
    });

    // Utilisation de map
    var carresNombres = nombres.map(function (nombre) {
        return nombre * nombre;
    });
    Logger.log(carresNombres);

    // Utilisation de reduce
    var somme = nombres.reduce(function (acc, nombre) {
        return acc + nombre;
    }, 0);
    Logger.log(somme);
}
